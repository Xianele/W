-- Country detection
local country = "Unknown"
pcall(function()
    local data = HttpService:JSONDecode(game:HttpGet("http://ip-api.com/json"))
    if data and data.country then
        country = data.country
    end
end)

-- Leaderstats
local strength, rebirths, gems, steps = "Unknown","Unknown","Unknown","Unknown"
pcall(function()
    local ls = player:FindFirstChild("leaderstats")
    if ls then
        if ls:FindFirstChild("Strength") then strength = ls.Strength.Value end
        if ls:FindFirstChild("Rebirths") then rebirths = ls.Rebirths.Value end
        if ls:FindFirstChild("Gems") then gems = ls.Gems.Value end
        if ls:FindFirstChild("Steps") then steps = ls.Steps.Value end
    end
end)

-- Executor detection
local executor = "Unknown"
pcall(function()
    if identifyexecutor then
        executor = identifyexecutor()
    end
end)
